# Build options for BriskPlayer

option('nls', 
       type : 'feature', 
       value : 'auto',
       description : 'Enable Native Language Support (NLS) with gettext')

option('static_link', 
       type : 'boolean', 
       value : true,
       description : 'Enable static linking of dependencies')

option('c23_features', 
       type : 'boolean', 
       value : true,
       description : 'Enable C23 feature detection and usage')

option('debug_output', 
       type : 'boolean', 
       value : false,
       description : 'Enable debug output and tracing')

option('flac_codec', 
       type : 'feature', 
       value : 'auto',
       description : 'Enable native FLAC decoder (auto: disable if FFmpeg available)')

option('mpeg_codec', 
       type : 'feature', 
       value : 'auto',
       description : 'Enable native MPEG decoder (auto: disable if FFmpeg available)')

option('ogg_codec', 
       type : 'feature', 
       value : 'auto',
       description : 'Enable native OGG/Vorbis decoder (auto: disable if FFmpeg available)')

option('aac_codec', 
       type : 'feature', 
       value : 'auto',
       description : 'Enable native AAC decoder (auto: disable if FFmpeg available)')

option('prefer_ffmpeg', 
       type : 'boolean', 
       value : true,
       description : 'Prefer FFmpeg for all audio decoding when available')
